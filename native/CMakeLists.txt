cmake_minimum_required(VERSION 3.16)
project(paos_g1d_sdk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(UNITREE_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../unitree_sdk2-main" CACHE PATH "Unitree SDK2 checkout")
set(BUILD_EXAMPLES OFF CACHE BOOL "Build SDK examples" FORCE)
add_subdirectory("${UNITREE_SDK_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/sdk" EXCLUDE_FROM_ALL)
add_library(g1d_sdk SHARED g1d_sdk.cpp)
target_link_libraries(g1d_sdk PRIVATE unitree_sdk2)
target_compile_options(g1d_sdk PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(g1d_sdk PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
install(TARGETS g1d_sdk LIBRARY DESTINATION lib)
enable_testing()
add_executable(test_joint_motion test_joint_motion.cpp)
target_compile_options(test_joint_motion PRIVATE -UNDEBUG)
add_test(NAME parameterized_joint_trajectory COMMAND test_joint_motion)
